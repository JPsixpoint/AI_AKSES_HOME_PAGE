// STEP 1: Set config values
const HEYGEN_API_KEY = 'YzEwZmEyOWJmYjdlNGI0ZWE3MzFiMjUzZWUzMzZiNTQtMTc0NjU4NDc4Nw=='; // Your HeyGen API Key
const AVATAR_ID = 'Sophie_public'; // HeyGen avatar
const VOICE_ID = 'c8e176c17f814004885fd590e03ff99f'; // HeyGen voice

// STEP 2: Create HeyGen session for avatar speaking
async function createHeyGenSession() {
  const res = await fetch('https://api.heygen.com/v1/streaming.new', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${HEYGEN_API_KEY}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      quality: 'high',
      avatar_id: AVATAR_ID,
      voice: {
        voice_id: VOICE_ID,
        rate: 1.0,
        emotion: 'Friendly'
      },
      disable_idle_timeout: true
    })
  });
  const data = await res.json();
  return data.data; // session_id, token, ws_url
}

// STEP 3: Send GPT (or AKSES) response to avatar to speak
async function speakWithHeyGen(session_id, token, text) {
  await fetch('https://api.heygen.com/v1/streaming.task', {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${token}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      session_id,
      text,
      task_type: 'repeat',
      task_mode: 'sync'
    })
  });
}

// STEP 4: Render the video stream above chat UI
function attachVideoStream(ws_url) {
  const socket = new WebSocket(ws_url);
  const video = document.getElementById('avatarVideo');

  socket.onmessage = (event) => {
    const msg = JSON.parse(event.data);
    if (msg.event === 'stream') {
      const streamUrl = msg.data?.url;
      if (streamUrl) {
        video.src = streamUrl;
        video.play();
      }
    }
  };
}

// STEP 5: Main function to use in your chat system
// Call this AFTER your LLM/Akses system sends the final message
async function runAvatarStream(responseText) {
  const { session_id, token, ws_url } = await createHeyGenSession();
  attachVideoStream(ws_url);
  await speakWithHeyGen(session_id, token, responseText);
}
